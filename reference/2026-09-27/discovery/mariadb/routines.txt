SELECT
 ROUTINE_SCHEMA,
 ROUTINE_NAME,
 ROUTINE_TYPE,
 DEFINER,
 SECURITY_TYPE
FROM information_schema.ROUTINES
WHERE ROUTINE_SCHEMA IN ('syslogdb','netlog_manager')
ORDER BY ROUTINE_SCHEMA,ROUTINE_NAME;
